```Go
    /*
```

## 4. 定义sturct
```Go
    /*
    8. 可以存放多种类型的结构struct, 类似于Class,还是枚举?
        type关键字定义struct, 然后是结构名, 然后再接一个struct关键字
    */
    type Point struct{
        //里面可以定义成员变量
        X int
        Y string
    }
    // 创建实例
    p := Point{1, 2}
    //修改变量
    p.X = 3

    /*
    9. 指针
        &取引用 *解引用(反引用)
    */
    p := Point{1, 2}
    q := p //只是在q里保存了p的当前值
    p.X = 3
    fmt.Println(p) //{3,2}
    fmt.Println(q) //{1,2} q没有变化, p的修改不影响q

    // 若想让pq保持同步
    q := &p //让q指向p的内存地址
    p.X = 3
    fmt.Println(*q) //{3,2} *表示反引用

    //修改q
    q.X = 10
    fmt.Println(*q) //{10,2} 

    /*
    10. 在方法中使用结构
    */
    //该方法定义在结构外
    func (p *Point) SetPoint(x,y,int){
        p.X = x
        p.y = y
```

## 5. 定义接口
```Go
    /*
    11. 接口
    */
    type ShapeInter interface {
        Print() //接口里的方法不需要func关键字和方法体
    }
    //如果一个struct想实现接口, 只要定义同名的方法即可
    type Rectangle struct{}
    type Tritangle struct{}
    func(r Rectangle) Print(){
        fmt.Println("矩形")
    }
    func(t Tritangle) Print(){
        fmt.Println("三角形")
    }
    //在main中
    var s Shape //定义接口变量
    s = Rectangle{} //给接口变量一个初始类型
    s.Println()// 矩形
    s = Tritangle{} //接口变量可以改类型
    s.Println()// 三角形

    /*
    接口作为函数参数使用
    */
    func printShape(s Shape){
        s.Print()
    }
    func main(){
        printShape(Rectangle{})
        printShape(Tritangle{})
    }


```

# 并发与多线程
1. Goroutines
```Go
func main(){
    go func2() //使用go关键字创建一个新的线程
    func1() //func1 和2会同时运行, 但是func1执行完毕后, main函数会立即关闭
}

```
2. 并发 Channels, 使各个线程相互通讯,共享变量
```Go
func main(){
    // 1. 使用chan关键字和make函数创建Channel, string是要传递的数据类型
    ch := make(chan string)
    // 发送和接收都会阻塞线程, 需要放在单独的Goroutines里
    go func2(ch)
    //3. 接受管道数据, 管道在右
    rest1 := <-ch

    go func1(ch)
    rest2 := <-ch
}
func func1(ch chan string){
    //2. 给管道发送数据, 管道在左, 数据在右
    ch <- "函数1"
}
func func2(ch chan string){
    ch <- "函数2"
}
```
