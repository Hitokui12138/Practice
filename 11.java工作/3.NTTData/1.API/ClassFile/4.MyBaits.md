## Mapper
```xml
<!-- DaoClass -->
<mapper namespace="com.XXX.dao.XXXDao">
    <!-- 1. Entity <-> resultMap -->
    <resultMap id="AAAMap"
      type="com.XXX.entity.AAAEntity">
        <result property="XXXNumber" column="xxx_number">
    </resultMap>

    <!-- 2. SELECT List -->
    <!-- Type可以是 "String" "integer"-->
    <select id="daoMethodName"
      resultType="com.XXX.entity.AAAEntity"
      resultMap="AAAMap">    <!-- 可选 -->
        SELECT...
        WHERE
          id = #{aaaEntity.id},
          number = #{number};
    </select>

    <!-- 3. INSERT -->
    <insert id="daoMethodName">
        INSERT INTO (...)
        VALUE(
            CURRENT_TIMESTAMP()
        );;
    </insert>

    <!-- 4. UPDATE 无条件更新一些, 并且有条件更新另一些-->
    <update id="daoMethodName">
        UPDATEE xxxTable
            <set>
                <!-- 无条件更新xxx_flg -->
                xxx_flg=#{XXXFlG}
                <!-- 有条件更新xxx_code -->
                <if test='isXXX == "1"'><!-- isXXX 是传进来的参数 -->
                    xxx_code + #{inDto.xxxCode}
                </if>
            </set>
        WHERE ...;
    </update>

```

## 参数是List的情况
1. 用于批量更新 List<XXList>,与IN结合使用
    ```xml
    <!-- 取出XXList中的每一个XXNumber用于充当IN()的参数 -->
    UPDATE
        xxxTBL
    SET
        XXFlg = '1'
    WHERE
        xxNumber IN(
            <foreach collection="xxEntityList"
                index="index"
                item="xxEntity" //只是个临时变量
                open=""
                separator=","
                close="">
                    #{xxEntity.xxNumber}
            </forrach>
        )
    ;
    ```
