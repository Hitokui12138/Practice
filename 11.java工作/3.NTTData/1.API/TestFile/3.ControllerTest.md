```java
            }
        }

```
## ControllerTest
1. 注解
    ```java
    @SpringBootTest
    @TestPropertySource(properties = {...})
    @AutoConfigureMockMvc //启动mockito
    class XXControllerTest extends BaseTests{...}
    ```
2. 成员变量
    ```java
    //模拟API的核心
    @Autowired
    private MockMvc mockMvc;

    @SpyBean
    controller;
    @SpyBean
    service;

    private static final String URL = "/xxx/xxx/apply"

    // 两种输入数据
    String jsonRequest = "";
    MultiValueMap<String, String> testInput = null;

    //@BeforeEach初始化InDto
    @BeforeEach
    public void beforeEach(){
        jsonRequest = getJsonString("request");
        testInput = convertJsonToMultiValueMap("request");
    }

    //有外部类时需要重制mock
    @AfterEach
    public void afterEach(){
        reset(xxService);
        s3Del("xxx");
    }
    ```

3. 正常终了
    ```java
    @Test
    @DisplayName("xxx")
    void test001(){
        //1. 先不管Service
        doNothing().when(xxxservice).execute(any());
        //2. 执行API
        mockMvc.perform(
            MockMvcRequestBuilders
                .post(URL)
                .content(jsonRequest)   //requestBody
                .contentType(MediaType.APPLICATION_JSON)
                .header("X-TRACEID", "xxx")   //requestHeader
        ).andExpect(status().is(200));// 判断返回结果
    }
    ```
4. 异常终了
    1. 参数校验
        ```java
        @ParameteriedTest(name="{0}")
        @DisplayName("xxx")
        @CsvSource({
            "'XX号码 null', xxNumber,",
            "'XX号码 数字Check', xxNumber,1234abc"
        })
        void test002(String name, String param, String value){
            //修改InputData的值
            testInput(param, value);

            //把Map转为JSON格式的String
            Map<String, String> map = testInput.toSingleValueMap();
            ObjectMapper mapper = new ObjectMapper();
            jsonRequest = objectMapper.writeValueAsString(map);

            //2. 执行API
        ```
