        ```bash
        cur_dir=`pwd`
        docker stop imooc-mysql
        docker rm imooc-mysql

        docker run --name imooc-mysql \
        -v ${cur_dir}/conf:/etc/mysql/conf.d \
        -v ${cur_dir}/conf:/var/lib/mysql \
        -p 3307:3306 \
        -e MYSQL_ROOT_PASSWORD=970601 \
        -d mysql:8.4  \
        --character-set-server=utf8mb4 \
        --collation-server=utf8mb4_unicode_ci
        ```

2. 启动ThriftServer的方法
    ```java
    @Configuration
    public class ThriftServer {
        @Value("${service.port}")
        private int servicePort;
        @Autowired
        private UserService.Iface userService;

        @PostConstruct //构造完成后立即启动
        public void startThriftServer(){
            // 初始化一个处理器
            TProcessor processor = new UserService.Processor<>(userService);
            // 创造一个NIO的Socket
            TNonblockingServerSocket socket = null;
            try{
                socket = new TNonblockingServerSocket(servicePort);
            } catch (TTransportException e){
                e.printStackTrace();
            }

            //构建server启动所需的参数
            TNonblockingServer.Args args = new TNonblockingServer.Args(socket);
            args.processor(processor);
            args.transportFactory(new TFramedTransport.Factory());//传输方式 帧传输
            args.protocolFactory(new TBinaryProtocol.Factory());//传输协议 二进制传输

            //构建Server
            TServer server = new TNonblockingServer(args);
            server.serve();
        }
    }
    ```
