#
```bash
brew update
brew install thrift
thrift -version

# demo
vi demo.thrift

# demo.thrift
## 1.namespace 表示要打包的语言
namespace java com.imooc.thrift.demo
namespace py thrift.demo
## 2.定义一个方法
service DemoService {
  void sayHello(1:string userName);
}

:wq


# 生成对应语言的文件
thrift --gen java demo.thrift
thrift --gen py demo.thrift


```
